function woodFill(tone) {
  return {
    background: [
      'repeating-linear-gradient(0deg,' + tint(tone.b, 24) + ' 0 1px, transparent 1px 6px)',
      'repeating-linear-gradient(0deg,' + tint(tone.b, 13) + ' 0 2px, transparent 2px 15px)',
      'repeating-linear-gradient(0deg, transparent 0 37px,' + tint(tone.b, 32) + ' 37px 39px, transparent 39px 96px)',
      'linear-gradient(180deg,' + tone.a + ',' + mix(tone.a, tone.b, 90) + ')',
    ].join(','),
  };
}

/* Fotóhely faerezetes lapon — a feliratos tábla jelzi, mi kerül ide. */
function WoodPlate({ t, label, ratio = '4 / 3', tone = 'tolgy', style, className }) {
  return (
    <div className={className} style={{ aspectRatio: ratio, border: '1px solid ' + t.line, display: 'flex', alignItems: 'center', justifyContent: 'center', overflow: 'hidden', ...woodFill(WOOD_TONES[tone]), ...style }}>
      <span style={{ fontFamily: t.f.m, fontSize: 10, letterSpacing: '.12em', textTransform: 'uppercase', color: t.ink, background: mix(t.bg, 'transparent', 90), padding: '5px 9px', border: '1px solid ' + t.line, textAlign: 'center', lineHeight: 1.4 }}>{label}</span>
    </div>
  );
}

/* Fotóhely faerezetes lapon — illusztrációs módban mért rajz lép a helyére. */
function WorkArt({ t, gfx, label, ratio = '4 / 3', tone = 'tolgy', motif = 'plan', icon, glyph, style, className }) {
  if (gfx === 'illu') return <Illu t={t} motif={motif} icon={icon} ratio={ratio} glyph={glyph} className={className} style={{ background: t.bg, ...style }} />;
  return <WoodPlate t={t} label={label} ratio={ratio} tone={tone} className={className} style={style} />;
}
